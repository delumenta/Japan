/* ========================================================= UPDATE FOOTER ARTWORK Active nav: use wide illustrated image Inactive nav: use circular image ========================================================= */ function updateFooterNavArtwork(){ document .querySelectorAll(".bottom-nav a, .bottom-nav button") .forEach(item => { const img = item.querySelector(".nav-icon img"); if(!img) return; if(item.classList.contains("active")){ const activeSrc = img.dataset.activeSrc; if(activeSrc){ img.src = activeSrc; } }else{ const normalSrc = img.dataset.normalSrc; if(normalSrc){ img.src = normalSrc; } } }); } /* Run AFTER footer.js has decided the active page */ updateFooterNavArtwork();
